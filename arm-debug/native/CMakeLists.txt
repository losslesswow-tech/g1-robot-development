cmake_minimum_required(VERSION 3.10)
project(g1_arm_debug LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(G1_ARM_DEBUG_ENABLE_CONTROL "Enable manually acquired, bounded arm sessions" OFF)
find_package(unitree_sdk2 REQUIRED CONFIG)
add_executable(g1_arm_debug worker.cpp)
target_link_libraries(g1_arm_debug unitree_sdk2)
target_compile_options(g1_arm_debug PRIVATE -Wall -Wextra -Wpedantic)
if(G1_ARM_DEBUG_ENABLE_CONTROL)
  target_compile_definitions(g1_arm_debug PRIVATE G1_ARM_DEBUG_ENABLE_CONTROL=1)
endif()
