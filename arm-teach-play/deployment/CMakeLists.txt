cmake_minimum_required(VERSION 3.10)
project(g1_arm_teach_play_deployment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(unitree_sdk2 REQUIRED CONFIG)
if(NOT DEFINED UPSTREAM_SOURCE)
  message(FATAL_ERROR "Specify the pinned UPSTREAM_SOURCE directory")
endif()
foreach(name teach play teach_with_name play_with_motions watchdog tts_probe)
  add_executable(${name} "${UPSTREAM_SOURCE}/${name}.cpp")
  target_link_libraries(${name} PRIVATE unitree_sdk2 pthread)
endforeach()
