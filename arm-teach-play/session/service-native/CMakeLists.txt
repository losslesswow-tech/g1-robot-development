cmake_minimum_required(VERSION 3.10)
project(g1_builtin_arm_service LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(unitree_sdk2 REQUIRED CONFIG)
add_executable(g1_builtin_arm_service service_control.cpp)
target_link_libraries(g1_builtin_arm_service PRIVATE unitree_sdk2 pthread)
target_compile_options(g1_builtin_arm_service PRIVATE -Wall -Wextra -Wpedantic)
