cmake_minimum_required(VERSION 3.10)
project(g1_network_modes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(unitree_sdk2 REQUIRED CONFIG)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs geometry_msgs std_msgs std_srvs topic_tools)
find_package(Threads REQUIRED)
add_executable(g1_network_bridge bridge.cpp)
target_include_directories(g1_network_bridge PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(g1_network_bridge unitree_sdk2 ${catkin_LIBRARIES} Threads::Threads rt)
add_executable(test_guard test_guard.cpp)
# Assertions must remain active in release builds.
target_compile_options(test_guard PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME guarded_motion_policy COMMAND test_guard)
