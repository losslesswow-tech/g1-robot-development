cmake_minimum_required(VERSION 3.10)
project(g1_bounded_arm LANGUAGES CXX)
option(G1_ENABLE_APPROVED_TRIAL "Compile physical trial entry point (requires separate authorization)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(unitree_sdk2 REQUIRED CONFIG)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs)
add_executable(g1_right_arm_bounded g1_right_arm_bounded.cpp)
if(G1_ENABLE_APPROVED_TRIAL)
  target_compile_definitions(g1_right_arm_bounded PRIVATE G1_ENABLE_APPROVED_TRIAL=1)
endif()
target_include_directories(g1_right_arm_bounded PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(g1_right_arm_bounded unitree_sdk2 ${catkin_LIBRARIES})
target_compile_options(g1_right_arm_bounded PRIVATE -Wall -Wextra -Wpedantic)
add_executable(g1_arm_service g1_arm_service.cpp)
target_link_libraries(g1_arm_service unitree_sdk2)
target_compile_options(g1_arm_service PRIVATE -Wall -Wextra -Wpedantic)
