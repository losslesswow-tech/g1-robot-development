cmake_minimum_required(VERSION 3.10)
project(g1_arm_readonly LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(unitree_sdk2 REQUIRED CONFIG)
add_executable(g1_arm_state g1_arm_state.cpp)
target_compile_options(g1_arm_state PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(g1_arm_state unitree_sdk2)
add_executable(g1_mode_query g1_mode_query.cpp)
target_compile_options(g1_mode_query PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(g1_mode_query unitree_sdk2)
